<script>
const filesEl=document.getElementById('files');
filesEl.onchange=()=>document.getElementById('fileCount').textContent=
  filesEl.files.length ? `${filesEl.files.length} resume(s) selected` : 'No files selected';

async function health(){
 try{
   const r=await fetch('/health');
   if(!r.ok) throw new Error();
   document.getElementById('status').textContent='Backend connected ✓';
   document.getElementById('status').className='status ok';
 }catch(e){
   document.getElementById('status').textContent='Backend not running. Start: uvicorn app:app --reload';
   document.getElementById('status').className='status bad';
 }
}
health();

document.getElementById('run').onclick=async()=>{
 const jd=document.getElementById('jd').value.trim();
 const files=filesEl.files;
 if(!jd){alert('Please enter the Job Description.');return}
 if(!files.length){alert('Please select at least one PDF/DOCX/TXT resume.');return}

 const fd=new FormData(); fd.append('jd',jd);
 for(const f of files) fd.append('resumes',f);

 document.getElementById('run').disabled=true;
 document.getElementById('status').textContent='Uploading and processing...';
 document.getElementById('status').className='status';

 try{
   const r=await fetch('/screen',{method:'POST',body:fd});
   if(!r.ok) throw new Error(await r.text());
   const d=await r.json();

   document.getElementById('result').style.display='block';
   document.getElementById('summary').textContent=`${d.results.length} candidate(s) ranked`;

   document.getElementById('rows').innerHTML=d.results.map(x=>`<tr>
   <td>${x.rank}</td>
   <td><b>${escapeHtml(x.candidate)}</b></td>
   <td class="score">${x.score}%</td>
   <td class="ok">${(x.matched_skills||[]).map(s=>`<span class="pill">${escapeHtml(s)}</span>`).join('')||'—'}</td>
   <td class="bad">${(x.missing_skills||[]).map(s=>`<span class="pill missing">${escapeHtml(s)}</span>`).join('')||'None'}</td>
   </tr>`).join('');

   document.getElementById('status').textContent='Done ✓';
   document.getElementById('status').className='status ok';
 }catch(e){
   document.getElementById('status').textContent='Upload failed: '+e.message;
   document.getElementById('status').className='status bad';
 }finally{
   document.getElementById('run').disabled=false;
 }
};

function escapeHtml(s){
 return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
